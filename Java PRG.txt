private static class PasswordUtil {
        private static final int ITERATIONS = 120000
private static String hash(String password) throws Exception {
            byte[] salt = new byte[16];
            new SecureRandom().nextBytes(salt);
            byte[] result = derive(password, salt, ITERATIONS);
            return "PBKDF2$" + ITERATIONS + "$" +
                    Base64.getEncoder().encodeToString(salt) + "$" +
                    Base64.getEncoder().encodeToString(result);
        }

        private static byte[] derive(String password, byte[] salt, int iterations) throws Exception {
            javax.crypto.spec.PBEKeySpec spec =
                    new javax.crypto.spec.PBEKeySpec(password.toCharArray(), salt, iterations, 256);
            try {
                return javax.crypto.SecretKeyFactory.getInstance("PBKDF2WithHmacSHA256")
                        .generateSecret(spec).getEncoded();
private static boolean verify(String password, String stored) throws Exception {
            if (stored == null || !stored.startsWith("PBKDF2$")) return password.equals(stored);
            String[] parts = stored.split("\\$");
            if (parts.length != 4) return false;
            byte[] salt = Base64.getDecoder().decode(parts[2]);
            byte[] expected = Base64.getDecoder().decode(parts[3]);
            byte[] actual = derive(password, salt, Integer.parseInt(parts[1]));
            return MessageDigest.isEqual(expected, actual);
        }
    }

        private BigDecimal percent(String value) {
            if(value.trim().isEmpty())return BigDecimal.ZERO;
            BigDecimal n=new BigDecimal(value.trim());
            if(n.signum()<0||n.compareTo(new BigDecimal("100"))>0)thrownew IllegalArgumentException("Discount and tax must be between 0 and 100.");
            return n;
        }
        private BigDecimal grandTotal() {
            BigDecimal d=percent(discount.getText()), t=percent(tax.getText());
            BigDecimal discountAmount=subtotal.multiply(d).divide(new BigDecimal("100"),2,java.math.RoundingMode.HALF_UP);
            BigDecimal after=subtotal.subtract(discountAmount);
 BigDecimal taxAmount=after.multiply(t).divide(new 
BigDecimal("100"),2,java.math.RoundingMode.HALF_UP);
